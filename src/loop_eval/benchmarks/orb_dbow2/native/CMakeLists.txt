cmake_minimum_required(VERSION 3.16)
project(ldo_orb_dbow2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(OpenCV 4 REQUIRED)
find_package(Eigen3 REQUIRED)
if(NOT ORB_SOURCE)
  message(FATAL_ERROR "Pass -DORB_SOURCE=<pinned ORB_SLAM3 source>")
endif()
add_executable(orb_dbow2 main.cc ${ORB_SOURCE}/src/ORBextractor.cc)
target_include_directories(orb_dbow2 PRIVATE ${ORB_SOURCE} ${ORB_SOURCE}/include ${ORB_SOURCE}/Thirdparty/Sophus)
target_link_libraries(orb_dbow2 PRIVATE Eigen3::Eigen ${OpenCV_LIBS} ${ORB_SOURCE}/Thirdparty/DBoW2/lib/libDBoW2.so)
target_compile_options(orb_dbow2 PRIVATE -O3 -march=native)
# The reference is only a verification tool; the benchmark does not link SLAM.
add_executable(orb_reference reference.cc)
find_package(Pangolin REQUIRED PATHS /opt/loop-detector/pangolin-install/lib/cmake/Pangolin)
target_include_directories(orb_reference PRIVATE ${ORB_SOURCE} ${ORB_SOURCE}/include ${ORB_SOURCE}/include/CameraModels ${ORB_SOURCE}/Thirdparty/Sophus /opt/loop-detector/pangolin-install/include)
target_link_libraries(orb_reference PRIVATE Eigen3::Eigen ${OpenCV_LIBS} ${Pangolin_LIBRARIES} ${ORB_SOURCE}/lib/libORB_SLAM3.so ${ORB_SOURCE}/Thirdparty/DBoW2/lib/libDBoW2.so)
target_compile_options(orb_reference PRIVATE -O3 -march=native)
target_compile_definitions(orb_reference PRIVATE HAVE_GLEW HAVE_EIGEN _LINUX_)
